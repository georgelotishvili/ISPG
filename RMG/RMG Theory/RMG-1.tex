\section{Primitive Postulates}

\begin{postulate}[One base-medium]
All low-energy matter modes, rods, clocks, signals and geometric excitations are
collective modes of one base-medium $\mathcal M$.
\end{postulate}

\begin{postulate}[Positive physical energy]
Physical excitations of $\mathcal M$ have Hamiltonian bounded below:
\[
H-H_{\rm vac}\ge0.
\]
There is no fundamental phantom degree of freedom.
\end{postulate}

\begin{postulate}[Relational observability]
Coordinate labels are not observables.  Only relations between physical modes
are observables.
\end{postulate}

\begin{postulate}[Stable low-energy phase]
The base-medium has a stable low-energy phase in which particles, rods, clocks
and signals are well-defined operational objects.
\end{postulate}

\section{Operational Bridge}

This section is not the microscopic derivation.  It is the minimal bridge needed
to recover the weak gravitational sector.

\begin{assumption}[Universal operational factor]
In the slowly varying low-energy regime, all stable matter modes share one local
operational factor:
\[
s=s(x).
\]
Equivalently, all matter modes see one effective metric
$g_{\mu\nu}^{\rm eff}$.
\end{assumption}

Status:
\[
\boxed{
\text{working assumption in RMG-1; open theorem from }\mathcal M.
}
\]

\begin{assumption}[Reciprocal clock/rod response]
Local clocks and rods respond as:
\[
d\tau=sdt,
\qquad
d\ell=s^{-1}dx.
\]
\end{assumption}

\begin{result}[Square signal law]
If the local physical signal speed is $c=d\ell/d\tau$, then:
\[
c
=
\frac{s^{-1}dx}{sdt}
=
s^{-2}\frac{dx}{dt}.
\]
Therefore:
\[
\boxed{
\frac{c_{\rm coord}}{c}=s^2.
}
\]
\end{result}

\begin{assumption}[Additive compression and multiplicative response]
Weak compression potentials add:
\[
u_{12}=u_1+u_2.
\]
Process factors multiply:
\[
s_{12}=s_1s_2.
\]
\end{assumption}

\begin{result}[Exponential layering]
The continuous solution with weak-field normalization is:
\[
\boxed{
s=e^{-u}.
}
\]
\end{result}

\begin{result}[Weak operational metric]
The bridge metric is:
\[
\boxed{
ds^2=-e^{-2u}c^2dt^2+e^{2u}\delta_{ij}dx^idx^j.
}
\]
\end{result}

\section{Static Weak-Field Sector}

\begin{assumption}[Hamiltonian source]
In the weak static regime, the compression/lapse constraint is sourced by total
physical energy density $\rho_E$:
\[
\rho_E(\mathbf x)\ge0.
\]
\end{assumption}

The inertial mass of a localized source is:
\[
M
=
\frac{1}{c^2}\int d^3x\,\rho_E(\mathbf x).
\]

Use the static energy functional:
\[
\mathcal E[u]
=
\frac{c^4}{8\pi G}\int d^3x\,|\nabla u|^2
-
\int d^3x\,\rho_Eu.
\]

\begin{result}[Static source equation]
Varying $u$ gives:
\[
\boxed{
\nabla^2u
=
-
\frac{4\pi G}{c^4}\rho_E.
}
\]
\end{result}

With $u(\infty)=0$:
\[
u(\mathbf x)
=
\frac{G}{c^4}
\int d^3x'\,
\frac{\rho_E(\mathbf x')}{|\mathbf x-\mathbf x'|}.
\]

Far from the source:
\[
\boxed{
u(r)=\frac{GM}{rc^2}+O(r^{-2}).
}
\]

Thus the gravitational charge is inertial mass because the source is total
energy.

The low-energy dynamic description must be expressible in ADM variables:
\[
ds^2

-N^2c^2dt^2

\[

\]

\begin{target}[Scalar and vector sectors]
The scalar/compression variable and the vector/flow variable must be
constraint or near-zone variables at low energy, not independent vacuum
radiation modes.
\end{target}

Target form:
\[
\nabla^2u=-4\pi G\rho/c^2,
\]
\[
\nabla^2\mathcal A_i
=
-
\kappa_VG\mathcal P_i^T/c^3.
\]

\begin{target}[Tensor/shear radiation]
The only propagating vacuum gravitational modes at low energy must be:
\[
h_+,
\qquad
h_\times.
\]
They must have positive energy and propagation speed:
\[
c_{\rm GW}=c.
\]
\end{target}

Target wave equation:
\[
\left(
\frac{1}{c^2}\partial_t^2-\nabla^2
\right)h_{ij}^{TT}
=
\kappa_TGT_{ij}^{TT}/c^4.
\]

Vacuum:
\[
\left(
\frac{1}{c^2}\partial_t^2-\nabla^2
\right)h_{ij}^{TT}=0.
\]

\section{Pass/Fail Gates}

RMG passes as a core framework if it derives:
\[
S_{\rm EH}+S_{\rm matter}
\]
as the low-energy fixed point of $\mathcal M$, with positive physical
Hamiltonian, universal metric coupling, and only two vacuum tensor
gravitational degrees of freedom.

species-dependent metric coupling, extra scalar/vector radiation, wrong
gravitational-wave speed, negative-energy physical modes, or non-closing ADM
constraints.
\end{gate}
